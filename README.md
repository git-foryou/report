#include <stdio.h>

int main() {
    
    int N; // 정수형태 변수 N을 선언

    printf("모래시계의 높이(3 이상의 홀수)를 입력하세요: ");
    scanf_s("%d", &N); // scanf 함수를 이용해 입력을 받고, 입력받은 값을 N에 받아 들인다.

    for (int i = 0; i <= N / 2; i++) { // 정수형태로 선언된 i의 값을 0으로 초기화한다. (N / 2) 의 값이 i보다 크거나 같다는 가정 하에, 해당 for문 안에 있는 중괄호 문장들을 실행 후 i의 값에 1을 더하는 것을 for문 안의 조건이 거짓이 되기 전까지 반복한다.
        for (int j = 0; j < i; j++) // 정수형태로 선언된 j의 값을 0으로 초기화한다. j가 위 for문에서 선언된 i의 값보다 작다는 가정 하에, 하위 문장을 실행 후 j의 값에 1을 더하는 것을  for문 안의 조건이 거짓되기 전까지 반복한다.
            printf(" "); // 띄어쓰기를 통해 점점 좁아지는 모양을 연출.
        for (int j = 0; j < N - 2 * i; j++) // 정수형태로 선언된 j의 값을 0으로 초기화한다. (N - 2 * i(상위 for문에 선언된 i 변수 참고)) 의 값이 J보다 크다는 가정 하에, 아래 문장 도달 후 j에 1을 더하는 것을 for문 안의 조건이 거짓이 되기 전까지 반복한다.
            printf("*"); // 역삼각형 모양을 연출하기 위한 문자.
        printf("\n"); // 그 다음 문단에 출력해야 하므로 \n을 이용해 줄바꿈한다.
    } 


    for (int i = N / 2 - 1; i >= 0; i--) { // 정수형태로 선언된 i의 값을 (N / 2 - 1) 로 초기화한다. i의 값이 0보다 크거나 같다는 가정 하에, i에 -1씩 빼는 것을 for문 안의 조건이 거짓이 되기 전까지 반복한다.
        for (int j = 0; j < i; j++) // 정수형태로 선언된 j의 값을 0으로 초기화한다. j가 위 for문에서 선언된 i값보다 작다는 가정 하에, 하위 문장 실행 후 j의 값에 1을 더하는 것을 for문 안의 조건이 거짓이 되기 전까지 반복한다.
            printf(" "); // 띄어쓰기를 통해 점점 좁아지는 모양을 연출
        for (int j = 0; j < N - 2 * i; j++) // 정수형태로 선언된 j의 값을 0으로 초기화한다. 이후 j의 값이 (N - 2 * i(상위 for문에 선언된 i 변수 참고)) 의 값이 j 보다 크다는 가정 하에, 아래 문장 도달 후 j에 1을 더하는 것을 for문 안의 조건이 거짓이 되기 전까지 반복한다.
            printf("*"); // 삼각형 모양을 연출하기 위한 문자. 
        printf("\n"); // 다음 문장에 모양을 연출하기 위해 \n 을 이용하여 줄바꿈한다. 
    }

    return 0;
}
